"use client";

import React, { useEffect, useState } from "react";
import Link from "next/link";
import { usePathname, useRouter } from "next/navigation";

interface UserProfile {
  username: string;
  role: "HEAD_AREA" | "UNIT";
  unit_name?: string;
  sentra_mikro?: string;
}

export function HeaderNav() {
  const pathname = usePathname();
  const router = useRouter();
  const [bankName, setBankName] = useState<string>("BANK EMOK");
  const [user, setUser] = useState<UserProfile | null>(null);

  useEffect(() => {
    // Sync session user dari LocalStorage
    const savedUser = localStorage.getItem("app_user");
    if (savedUser) {
      try {
        setUser(JSON.parse(savedUser));
      } catch (e) {
        console.error("Failed to parse user session");
      }
    } else {
      // Default fallback jika belum login (dianggap HEAD_AREA agar tab tidak hilang saat dev)
      setUser({
        username: "Head Office",
        role: "HEAD_AREA",
      });
    }

    // Fetch Global Settings / Bank Name
    fetch("/api/reports")
      .then((res) => res.json())
      .then((data) => {
        if (data.bank_name) setBankName(data.bank_name);
      })
      .catch(() => {});
  }, []);

  const handleLogout = () => {
    localStorage.removeItem("app_user");
    router.push("/login");
  };

  const isHeadArea = !user || user.role === "HEAD_AREA";

  return (
    <header className="w-full bg-slate-950 border-b border-slate-800 text-slate-100 sticky top-0 z-40 shadow-xl">
      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 h-16 flex items-center justify-between">
        {/* Brand Logo & Title */}
        <div className="flex items-center gap-6">
          <Link href={isHeadArea ? "/head-area" : "/unit-execution"} className="flex items-center gap-2.5 group">
            <div className="w-9 h-9 rounded-lg bg-indigo-600 flex items-center justify-center font-black text-white text-sm shadow-lg shadow-indigo-600/30 group-hover:bg-indigo-500 transition-all">
              BK
            </div>
            <span className="font-extrabold tracking-wider text-base text-white group-hover:text-indigo-400 transition-colors uppercase">
              {bankName}
            </span>
          </Link>

          {/* Dynamic Nav Items Based On Role */}
          <nav className="hidden md:flex items-center gap-1.5 text-xs font-semibold">
            {isHeadArea ? (
              <>
                {/* 1. KONSOLIDASI UNIT */}
                <Link
                  href="/head-area"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/head-area"
                      ? "bg-indigo-600 text-white font-bold shadow-md shadow-indigo-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  KONSOLIDASI UNIT
                </Link>

                {/* 2. REKAP DATA EKSEKUSI */}
                <Link
                  href="/head-execution"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/head-execution"
                      ? "bg-indigo-600 text-white font-bold shadow-md shadow-indigo-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  REKAP DATA EKSEKUSI
                </Link>

                {/* 3. PENGATURAN */}
                <Link
                  href="/settings"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/settings"
                      ? "bg-amber-600 text-white font-bold shadow-md shadow-amber-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  ⚙️ PENGATURAN
                </Link>

                {/* 4. BRANGKAS DOKUMEN */}
                <Link
                  href="/vault"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/vault"
                      ? "bg-indigo-600 text-white font-bold shadow-md shadow-indigo-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  💼 BRANGKAS DOKUMEN
                </Link>
              </>
            ) : (
              <>
                {/* 1. DATA EKSEKUSI UNIT */}
                <Link
                  href="/unit-execution"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/unit-execution"
                      ? "bg-rose-600 text-white font-bold shadow-md shadow-rose-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  DATA EKSEKUSI
                </Link>

                {/* 2. MODUL KEPALA UNIT */}
                <Link
                  href="/unit-control"
                  className={`px-3 py-1.5 rounded-lg transition-all ${
                    pathname === "/unit-control"
                      ? "bg-indigo-600 text-white font-bold shadow-md shadow-indigo-600/30"
                      : "text-slate-400 hover:text-slate-200 hover:bg-slate-900"
                  }`}
                >
                  MODUL KEPALA UNIT
                </Link>
              </>
            )}
          </nav>
        </div>

        {/* User Badge & Switcher / Logout */}
        <div className="flex items-center gap-3">
          <div className="text-right hidden sm:block">
            <div className="text-xs font-bold text-slate-200">
              {user?.username || "Head Area"}{" "}
              <span className="text-slate-400 font-normal">({user?.role || "HEAD_AREA"})</span>
            </div>
            <div className="text-[10px] text-slate-400 font-mono">
              {user?.unit_name ? `${user.unit_name} (${user.sentra_mikro})` : "Head Office Area"}
            </div>
          </div>

          <span
            className={`px-2 py-0.5 rounded text-[10px] font-bold font-mono uppercase ${
              isHeadArea
                ? "bg-indigo-950 text-indigo-400 border border-indigo-800"
                : "bg-emerald-950 text-emerald-400 border border-emerald-800"
            }`}
          >
            {user?.role || "HEAD_AREA"}
          </span>

          <button
            onClick={handleLogout}
            className="px-3 py-1.5 bg-slate-900 hover:bg-slate-800 border border-slate-800 text-slate-300 hover:text-white rounded-lg text-xs font-semibold transition-all cursor-pointer"
          >
            Keluar
          </button>
        </div>
      </div>
    </header>
  );
}
